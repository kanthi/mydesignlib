Developer Infrastructure

Full programmatic control in three lines of code.

Deploy state-driven video pipelines across web applications, video editors, and game engines using our open Python SDK or REST endpoints. 

import modalis

# 1. Initialize client with open-weights or hosted endpoint
client = modalis.Client(api_key="sk_live_modalis_world_model")

# 2. Stage base master keyframe and condition state change
transition = client.states.create_transition(
    base_frame="assets/state-base.png",
    dimension="scene",
    target_prompt="Pale blue architectural walls translate outward, revealing sky",
    camera_lock=True,
    duration_seconds=2.08,
    terminal_guard_seconds=0.08
)

# 3. Stream paired forward and reverse video buffers
transition.export_pair(output_dir="./exports/", format="mp4")
print(f"Ready: {transition.forward_url} <-> {transition.reverse_url}")
