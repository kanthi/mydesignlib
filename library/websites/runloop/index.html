Developer First Integration

Drop in 4 lines of code to enable automatic routing, tool firewalls, and observability.

import { Runloop } from '@runloop/sdk'; const rl = new Runloop({ apiKey: process.env.RUNLOOP_KEY }); // Spawn routed agent session with automatic tool call verification const session = await rl.agents.create({ model: 'auto/frontier-fast', guardrails: { strictJson: true, piiMask: true }, tools: [rl.tools.codeSandbox(), rl.tools.webSearch()] }); const stream = await session.run({ prompt: 'Refactor auth handler to edge' }); for await (const chunk of stream) { process.stdout.write(chunk); }
